CLI 管理
guardianctl status     # 进程状态+CPU/内存+模块清单
guardianctl health     # 4项健康检查（Python组件/保护模块/eBPF/配置）
guardianctl test       # 全组件自检
guardianctl stats      # 最近1小时告警统计
guardianctl dump       # 导出基线模型
guardianctl logs -f    # 实时日志

systemd 守护
Restart=on-failure     # 崩溃自动重启
WatchdogSec=30s        # 系统级看门狗
MemoryMax=800M         # 硬内存限制
CPUQuota=200%          # CPU 限制
ProtectSystem=strict   # 安全加固
